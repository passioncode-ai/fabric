const { app, BrowserWindow } = require('electron'); const fs = require('fs'); const path = require('path')
const [, html, out, w = '1000', h = '680'] = process.argv.slice(1)
app.whenReady().then(async () => {
  const win = new BrowserWindow({ width: +w, height: +h, show: false, webPreferences: { preload: path.join(__dirname, 'preload.cjs'), contextIsolation: false, sandbox: false, offscreen: true } })
  await win.loadFile(html)
  await new Promise((r) => setTimeout(r, 2500))
  fs.writeFileSync(out, (await win.webContents.capturePage()).toPNG())
  app.quit()
})
